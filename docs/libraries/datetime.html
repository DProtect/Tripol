Gregorian calendar calculations without time zones.

show datetime.days_in_month(2024, 2)
Important notes

Years 1..9999; no time zones or wall-clock API.

datetime.days_in_month #

Days in month

func datetime.days_in_month(y:int,m:int)->int
View implementation
func days_in_month(y:int,m:int)->int {
 assert y>=1 and y<=9999 and m>=1 and m<=12,"Invalid date"
 if m==2 {if leap(y) {return 29};return 28}
 if m==4 or m==6 or m==9 or m==11 {return 30}
 return 31
}

datetime.valid #

Validate date

func datetime.valid(y:int,m:int,d:int)->bool
View implementation
func valid(y:int,m:int,d:int)->bool {
 if y<1 or y>9999 or m<1 or m>12 or d<1 {return false}
 return d<=days_in_month(y,m)
}

datetime.day_of_year #

Day number within year

func datetime.day_of_year(y:int,m:int,d:int)->int
View implementation
func day_of_year(y:int,m:int,d:int)->int {
 assert valid(y,m,d),"Invalid date"
 total=d
 for i in range(1,m) {total+=days_in_month(y,i)}
 return total
}

datetime.ordinal #

Ordinal day since year 1

func datetime.ordinal(y:int,m:int,d:int)->int
View implementation
func ordinal(y:int,m:int,d:int)->int {
 assert valid(y,m,d),"Invalid date"
 n=y-1
 return n*365+n/4-n/100+n/400+day_of_year(y,m,d)
}
