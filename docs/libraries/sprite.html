func draw_key(dst:ints,w:int,h:int,src:ints,sw:int,sh:int,x:int,y:int,key:int)->int {
 assert len(dst)==canvas.size(w,h) and len(src)==canvas.size(sw,sh),"Invalid sprite shape"
 assert x>=-4096 and x<=4096 and y>=-4096 and y<=4096,"Sprite coordinate out of range"
 n=0
 for row in range(max(0,-y),max(0,min(sh,h-y))) {
  for col in range(max(0,-x),max(0,min(sw,w-x))) {
   c=src[row*sw+col];assert colors.valid(c),"Invalid sprite color"
   if c!=key {dst[(y+row)*w+x+col]=c;n+=1}
  }
 }
 return n
}
