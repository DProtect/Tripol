tripol.arrays

Integer array utilities and native copying.

ASCENSION 0.248 functions

Example

import tripol.arrays

values = arrays.make(4, 7)
show arrays.total(values)
copy = ints(4)
assert arrays.copy_into(copy, values)
show copy
Important notes

copy_into requires a mutable dynamic destination with exactly the source length. Fixed arrays can be borrowed for read-only operations. Integer overflow remains checked.

arrays.make #

Allocate and fill an array

func arrays.make(n:int, value:int) -> ints
View implementation
func make(n:int, value:int) -> ints {
    a=ints(n)
    for i in range(n) { a[i]=value }
    return a
}

arrays.count #

Count occurrences

func arrays.count(a:ints, value:int) -> int
View implementation
func count(a:ints, value:int) -> int {
    n=0
    for item in a { if item==value { n+=1 } }
    return n
}

arrays.reversed #

Return reversed copy

func arrays.reversed(a:ints) -> ints
View implementation
func reversed(a:ints) -> ints {
    b=ints(len(a))
    for i in range(len(a)) { b[i]=a[len(a)-1-i] }
    return b
}
