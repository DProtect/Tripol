tripol.camera

Integer camera coordinates and visibility.

ASCENSION 0.244 functions

Example

import tripol.camera

show camera.screen(120, 100, 2)
Important notes

Zoom is an integer from 1 to 16. world uses integer division.

camera.screen #

Convert world coordinate to screen coordinate

func camera.screen(world:int,origin:int,zoom=1)->int
View implementation
func screen(world:int,origin:int,zoom=1)->int {assert zoom>=1 and zoom<=16,"Invalid zoom";return (world-origin)*zoom}

camera.world #

Convert screen coordinate to world coordinate

func camera.world(screen_pos:int,origin:int,zoom=1)->int
View implementation
func world(screen_pos:int,origin:int,zoom=1)->int {assert zoom>=1 and zoom<=16,"Invalid zoom";return screen_pos/zoom+origin}

camera.follow #

Center and clamp camera origin

func camera.follow(pos:int,view:int,limit:int)->int
View implementation
func follow(pos:int,view:int,limit:int)->int {assert view>=0 and limit>=view,"Invalid camera bounds";return max(0,min(pos-view/2,limit-view))}

camera.visible #

Test rectangle intersection with viewport

func camera.visible(x:int,y:int,w:int,h:int,vw:int,vh:int)->bool
View implementation
func visible(x:int,y:int,w:int,h:int,vw:int,vh:int)->bool {return w>0 and h>0 and vw>0 and vh>0 and x<vw and y<vh and x+w>0 and y+h>0}
