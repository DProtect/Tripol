func length2(x:float,y:float)->float {return sqrt(dot2(x,y,x,y))}

View implementation
func normalized_x(x:float,y:float)->float {n=length2(x,y);assert n>0.0,"Zero vector";return x/n}

func scale(a:ints,k:int)->ints {b=ints(len(a));for i in range(len(a)) {b[i]=a[i]*k};return b}
