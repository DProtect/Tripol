bits.mask #

Compute mask for bit index 0..62

func bits.mask(index:int)->int
View implementation
func mask(index:int)->int {
 assert index>=0 and index<63,"Invalid bit index"
 n=1;repeat index {n*=2};return n
}

Test bit

func test(n:int,index:int)->bool {assert n>=0,"Negative bit value";return n/mask(index)%2==1}

Count set bits

func count(n:int)->int {
 assert n>=0,"Negative bit value";total=0
 while n>0 {total+=n%2;n/=2}
 return total
}
