tripol.image

RGB transformations and small ASCII PPM exports.

ASCENSION 0.246 functions

Example

import tripol.image

pixels = [16777215, 0]
show image.flip_x(pixels, 2, 1)
Important notes

PPM export is ASCII P3, max 256 pixels; no PNG or JPEG codec.

Interactive graphics preview

Choose Try it and Run. The engine compiles real code and displays frames from game.present or game.gl_frame. Uses WebGPU when available, otherwise Canvas 2D. Finite playback, not live input.

Demo 1

import tripol.game
import tripol.canvas
import tripol.image

w = 64
h = 48
game.window("Flipped image", w, h)
pixels = canvas.create(w, h)
canvas.clear(pixels, 1052688)
canvas.rectangle(pixels, w, h, 4, 8, 20, 32, 16753920)
flipped = image.flip_x(pixels, w, h)
game.present(flipped, w, h)
game.close()

Return horizontally flipped copy

View implementation
func flip_x(pixels:ints,w:int,h:int)->ints {
 assert len(pixels)==canvas.size(w,h),"Invalid image"
 b=ints(len(pixels));for y in range(h) {for x in range(w) {b[y*w+x]=pixels[y*w+w-1-x]}};return b
}

image.crop #

Copy rectangular crop

View implementation
func crop(pixels:ints,w:int,h:int,x:int,y:int,cw:int,ch:int)->ints {
 assert len(pixels)==canvas.size(w,h),"Invalid image"
 n=canvas.size(cw,ch);assert x>=0 and y>=0 and x<=w-cw and y<=h-ch,"Invalid crop"
 b=ints(n);for r in range(ch) {for c in range(cw) {b[r*cw+c]=pixels[(y+r)*w+x+c]}};return b
}

image.ppm #

Encode P3 PPM (max 256 pixels)

func image.ppm(pixels:ints,w:int,h:int)->string
View implementation
func ppm(pixels:ints,w:int,h:int)->string {
 assert len(pixels)==canvas.size(w,h) and len(pixels)<=256,"PPM R1 max 256 pixels"
 s="P3\n"+text(w)+" "+text(h)+"\n255\n"
 for c in pixels {s=s+text(colors.red(c))+" "+text(colors.green(c))+" "+text(colors.blue(c))+"\n"}
 return s
}
