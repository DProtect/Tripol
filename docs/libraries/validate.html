validate.integer #

Check integer text syntax

func validate.integer(s:string) -> bool
View implementation
func integer(s:string) -> bool {
    n=len(s); if n==0 {return false}
    i=0
    if byte_at(s,0)==43 or byte_at(s,0)==45 {i=1}
    if i==n {return false}
    while i<n {if not digit(byte_at(s,i)) {return false}; i+=1}
    return true
}

validate.identifier #

Check ASCII identifier syntax

func validate.identifier(s:string) -> bool
View implementation
func identifier(s:string) -> bool {
    if len(s)==0 {return false}
    b=byte_at(s,0); if not (letter(b) or b==95) {return false}
    for i in range(1,len(s)) {b=byte_at(s,i);if not (letter(b) or digit(b) or b==95) {return false}}
    return true
}
