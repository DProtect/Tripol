tripol.discord

Bots, callbacks and message builders. Gateway and REST on Windows and Linux.

DISCORD REVOLUTION 0.2860 functions
Where to start

Try a text command and an embed in the Playground. For a live bot on Windows, see the guide. DISCORD REVOLUTION

Example

import tripol.discord

bot = discord.bot()
bot.prefix("!")
bot.command("hola", "¡Hola, compañero!")
bot.command("ping", "Pong!")
show bot.preview("!hola")

embed = discord.embed()
embed.title("DISCORD REVOLUTION")
embed.description("Un embed de TRIPOL")
embed.color("#00A0FF")
embed.field("Versión", "0.28.0", true)
show embed.payload()
import tripol.discord

discord.reply("!hello", "Hello!")
show discord.preview("!hello")
Important notes

Windows and Linux x86-64 for live networking. Up to 16 exact commands. Enable Message Content Intent and channel permissions. Gateway replies can route to any permitted channel; say and welcome require a selected channel. Automatic reconnect/Resume and slash commands are not supported. Never publish a token or an executable containing it. identify_payload contains the token; never log it. HTTP failures can stop the program.

discord.run #

Start Windows Gateway loop

func discord.run(intents=33281) -> int
Source code
func run(intents=33281)->int {
 assert intents==1 or intents==33281,"Unsupported Discord Gateway intents"
 assert configured(),"Call discord.login first"
 scratch {assert has_channel() or len(bot_slot_get(36))==0,"Welcome requires a selected channel"}
 assert gw_open(),"Gateway TLS/WebSocket open failed (Windows only)"
 bot_slot_set(37,"");bot_slot_set(38,"")
 show "TRIPOL Discord Gateway connecting..."
 interval=0; sequence=-1; next_heartbeat=0; sent=0; awaiting=false; done=false
 started=time.now_ms(); ready=false
 while not done {
  scratch {
   // Wait on a local thread handle, not a REST request. Wakes on data arrival.
   event=gw_receive(100)
   if event=="!" {done=true;show "Gateway closed or receive failed."}
   else if len(event)>0 {
    op=errors.int_or(json.get_raw_large(event,"op"),-1)
    if op==10 and interval==0 {
     d=json.get_raw_large(event,"d")
     interval=errors.int_or(json.get_raw_large(d,"heartbeat_interval"),0)
     if interval<1000 or interval>300000 {gw_abort()}
     next_heartbeat=time.now_ms()+time.now_ns()%interval
     gw_send(identify_payload(intents))
    } else if op==0 {
     seq=errors.int_or(json.get_raw_large(event,"s"),-1)
     if seq>=0 {sequence=seq}
     kind=json.get_raw_large(event,"t")
     if kind=="\"READY\"" and not ready {
      ready=true;bot_slot_set(38,"ready")
      show "TRIPOL Discord Gateway READY - Ctrl+C to stop."
      greeting=bot_slot_get(36)
      if len(greeting)>0 {say(greeting)}
     } else if ready and kind=="\"MESSAGE_CREATE\"" {
      response=gateway_reply(event)
      if len(response)>0 {send_to(event_channel(event),response)}
     }
    } else if op==11 {
     if awaiting {bot_slot_set(37,text(time.now_ms()-sent));awaiting=false}
    } else if op==1 {
     gw_send(heartbeat_payload(sequence));sent=time.now_ms();awaiting=true
    } else if op==7 or op==9 {
     done=true;show "Gateway requested reconnect or invalidated the session. Stopped safely."
    }
   }
   if not done {
    now=time.now_ms()
    if not ready and now-started>30000 {gw_abort()}
    if interval>0 and now>=next_heartbeat {
     if awaiting {gw_abort()}
     gw_send(heartbeat_payload(sequence));sent=now;awaiting=true
     next_heartbeat=now+interval
    }
   }
  }
 }
 gw_close();bot_slot_set(38,"")
 return 1
}
