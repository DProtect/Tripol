Generate arithmetic sequence

View implementation
func arithmetic(n:int,start:int,step:int)->ints {
 a=ints(n);v=start
 for i in range(n) {a[i]=v;if i+1<n {v+=step}}
 return a
}

View implementation
func fibonacci(n:int)->ints {
 assert n>=0 and n<=93,"Fibonacci length 0..93"
 a=ints(n);if n>1 {a[1]=1}
 for i in range(2,n) {a[i]=a[i-1]+a[i-2]}
 return a
}
