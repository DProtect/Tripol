Documentation / 0.24

tripol.compression

Integer run-length encoding, not a general byte codec.

ASCENSION 0.243 functions

Example

import tripol.compression

items = [4, 4, 4, 9]
encoded = compression.rle_encode(items)
show compression.rle_decode(encoded)
Important notes

Decoded length is limited to 65,535. Encoded data uses [value, count, ...].

compression.rle_encode #

Encode value/count pairs

func compression.rle_encode(a:ints)->ints
View implementation
func rle_encode(a:ints)->ints {
 if len(a)==0 {return ints(0)}
 runs=1
 for i in range(1,len(a)) {if a[i]!=a[i-1] {runs+=1}}
 assert runs<=32767,"Too many RLE runs"
 b=ints(runs*2)
 j=0; b[0]=a[0]; b[1]=1
 for i in range(1,len(a)) {
  if a[i]==b[j] {b[j+1]=b[j+1]+1} else {j+=2;b[j]=a[i];b[j+1]=1}
 }
 return b
}

compression.rle_valid #

Validate run pairs and decoded size

func compression.rle_valid(a:ints)->bool
View implementation
func rle_valid(a:ints)->bool {
 if len(a)%2!=0 {return false}
 total=0
 for i in range(len(a)/2) {
  n=a[i*2+1]
  if n<=0 or n>65535-total {return false}
  total+=n
 }
 return true
}

compression.rle_decode #

Decode valid run pairs

func compression.rle_decode(a:ints)->ints
View implementation
func rle_decode(a:ints)->ints {
 assert rle_valid(a),"Invalid RLE"
 total=0
 for i in range(len(a)/2) {total+=a[i*2+1]}
 b=ints(total); j=0
 for i in range(len(a)/2) {repeat a[i*2+1] {b[j]=a[i*2];j+=1}}
 return b
}
