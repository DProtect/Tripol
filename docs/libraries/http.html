tripol.http

Minimal HTTP/1.0 text requests and response parsing.

ASCENSION 0.243 functions

Example

import tripol.http

response = "HTTP/1.0 200 OK\r\n\r\nHello"
show http.status(response)
show http.body(response)
Important notes

No TLS, chunk decoding or DNS in get_raw. Explicit IPv4 address and port. Response includes headers and body.

http.get_raw #

Send plain HTTP GET to explicit IPv4 address

func http.get_raw(address:int,port:int,host:string,path:string)->string
View implementation
func get_raw(address:int,port:int,host:string,path:string)->string {
 assert len(host)>0 and len(path)>0,"HTTP host and path required"
 assert byte_at(path,0)==47,"HTTP path must begin with /"
 for i in range(len(host)) {assert byte_at(host,i)>32 and byte_at(host,i)<127,"Invalid HTTP host"}
 for i in range(len(path)) {assert byte_at(path,i)>32 and byte_at(path,i)<127,"Invalid HTTP path"}
 return network.exchange(address,port,"GET "+path+" HTTP/1.0\r\nHost: "+host+"\r\nConnection: close\r\n\r\n")
}

http.status #

Parse HTTP status

func http.status(response:string)->int
View implementation
func status(response:string)->int {
 assert len(response)>=12,"Short HTTP response"
 assert has_prefix(response,"HTTP/1.0 ") or has_prefix(response,"HTTP/1.1 "),"Invalid HTTP version"
 a=byte_at(response,9)-48
 b=byte_at(response,10)-48
 c=byte_at(response,11)-48
 assert a>=1 and a<=5 and b>=0 and b<=9 and c>=0 and c<=9,"Invalid HTTP status"
 return a*100+b*10+c
}

http.body #

Return raw body after header separator (no chunk decoding)

func http.body(response:string)->string
View implementation
func body(response:string)->string {
 i=find_bytes(response,"\r\n\r\n")
 assert i>=0,"Missing HTTP header separator"
 return slice(response,i+4,len(response))
}
